// InitDatabase initializes the connection to NeonDB (PostgreSQL) using GORM.
func InitDatabase() (*gorm.DB, error) {
        dsn := os.Getenv("DATABASE_URL")
        if dsn == "" {
                return nil, fmt.Errorf("DATABASE_URL environment variable is not set")
        }

        db, err := gorm.Open(postgres.Open(dsn), &gorm.Config{})
        if err != nil {
                return nil, fmt.Errorf("failed to connect to database: %w", err)
        }

        sqlDB, err := db.DB()
        if err != nil {
                return nil, fmt.Errorf("failed to get database connection pool: %w", err)
        }

        sqlDB.SetMaxOpenConns(10)
        sqlDB.SetMaxIdleConns(5)
        sqlDB.SetConnMaxLifetime(30 * time.Minute)
        sqlDB.SetConnMaxIdleTime(5 * time.Minute)

        if err := sqlDB.Ping(); err != nil {
                return nil, fmt.Errorf("failed to ping database: %w", err)
        }

        slog.Info("Successfully connected to the database")

        if err := db.AutoMigrate(&models.User{}); err != nil {
                slog.Warn("Failed to auto-migrate User model", "error", err)
        }

        return db, nil
}

func (c *AuthController) setAuthCookies(w http.ResponseWriter, token, refreshToken string) {
        http.SetCookie(w, &http.Cookie{
                Name:     "access_token",
                Value:    token,
                Path:     "/",
                HttpOnly: true,
                Secure:   false,
                SameSite: http.SameSiteLaxMode,
                MaxAge:   900, // 15 minutes
        })
        http.SetCookie(w, &http.Cookie{
                Name:     "refresh_token",
                Value:    refreshToken,
                Path:     "/",
                HttpOnly: true,
                Secure:   false,
                SameSite: http.SameSiteLaxMode,
                MaxAge:   7 * 24 * 3600, // 7 days
        })
}

func (c *AuthController) clearAuthCookies(w http.ResponseWriter) {

func (c *AuthController) Register(w http.ResponseWriter, r *http.Request) {
        r.Body = http.MaxBytesReader(w, r.Body, 1<<20) // 1 MB max body
        var req dto.RegisterRequest
        if err := json.NewDecoder(r.Body).Decode(&req); err != nil {
                w.Header().Set("Content-Type", "application/json")
                w.WriteHeader(http.StatusBadRequest)
                json.NewEncoder(w).Encode(map[string]string{"error": "El formato de la solicitud de registro es inválido."})
                return
        }

        req.Email = strings.TrimSpace(req.Email)
        req.FirstName = strings.TrimSpace(req.FirstName)
        req.LastName = strings.TrimSpace(req.LastName)
        req.Phone = strings.TrimSpace(req.Phone)

        if err := c.validate.Struct(req); err != nil {
                w.Header().Set("Content-Type", "application/json")
                w.WriteHeader(http.StatusBadRequest)
                json.NewEncoder(w).Encode(map[string]string{"error": "Datos de registro inválidos: revise nombre, apellido, email, teléfono y contraseña (mínimo 8 caracteres, con mayúscula, minúscula y carácter especial)."})
                return
        }

        resp, err := c.authService.Register(r.Context(), &req)
        if err != nil {
                w.Header().Set("Content-Type", "application/json")
                w.WriteHeader(http.StatusBadRequest)
                json.NewEncoder(w).Encode(map[string]string{"error": err.Error()})
                return
        }

        w.Header().Set("Content-Type", "application/json")
        c.setAuthCookies(w, resp.Token, resp.RefreshToken)
        w.WriteHeader(http.StatusCreated)
        json.NewEncoder(w).Encode(resp)
}

// Login handles step 2: given a user_id and role_id (no password), issues the JWT token.
func (c *AuthController) Login(w http.ResponseWriter, r *http.Request) {
        r.Body = http.MaxBytesReader(w, r.Body, 1<<20) // 1 MB max body
        var req dto.LoginRequest
        if err := json.NewDecoder(r.Body).Decode(&req); err != nil {
                w.Header().Set("Content-Type", "application/json")
                w.WriteHeader(http.StatusBadRequest)
                json.NewEncoder(w).Encode(map[string]string{"error": "El formato de la solicitud de selección de rol es inválido."})
                return
        }

        if err := c.validate.Struct(req); err != nil {
                w.Header().Set("Content-Type", "application/json")
                w.WriteHeader(http.StatusBadRequest)
                json.NewEncoder(w).Encode(map[string]string{"error": "El usuario y el rol son obligatorios."})
                return
        }

        resp, err := c.authService.Login(r.Context(), &req)
        if err != nil {
                w.Header().Set("Content-Type", "application/json")
                w.WriteHeader(http.StatusUnauthorized)
                json.NewEncoder(w).Encode(map[string]string{"error": err.Error()})
                return
        }

        w.Header().Set("Content-Type", "application/json")
        if resp.UserStatus == "blocked" {
                w.WriteHeader(http.StatusOK)
                json.NewEncoder(w).Encode(resp)
                return
        }
        c.setAuthCookies(w, resp.Token, resp.RefreshToken)
        w.WriteHeader(http.StatusOK)
        json.NewEncoder(w).Encode(resp)
}

                http.Error(w, "unauthorized", http.StatusUnauthorized)
                return
        }

        r.Body = http.MaxBytesReader(w, r.Body, 1<<20)
        var req dto.LinkGoogleRequest
        if err := json.NewDecoder(r.Body).Decode(&req); err != nil {
                w.Header().Set("Content-Type", "application/json")
                w.WriteHeader(http.StatusBadRequest)
                json.NewEncoder(w).Encode(map[string]string{"error": "Cuerpo de solicitud inválido"})
                return
        }

        if err := c.authService.LinkGoogleAccount(r.Context(), userID, &req); err != nil {
                w.Header().Set("Content-Type", "application/json")
                w.WriteHeader(http.StatusBadRequest)
                json.NewEncoder(w).Encode(map[string]string{"error": err.Error()})
                return
        }

        w.Header().Set("Content-Type", "application/json")
        w.WriteHeader(http.StatusOK)
        json.NewEncoder(w).Encode(map[string]string{"message": "Cuenta de Google vinculada correctamente"})
}

// JWTService interface for validation, to avoid circular dependencies
type TokenValidatorJWTService interface {
        ValidateToken(tokenString string) (jwt.MapClaims, error)
}

type TokenValidatorController struct {
        jwtService TokenValidatorJWTService
}

func NewTokenValidatorController(jwtService TokenValidatorJWTService) *TokenValidatorController {
        return &TokenValidatorController{
                jwtService: jwtService,
        }
}

// ValidateToken is an internal endpoint called by NGINX auth_request
// It returns 200 OK and sets X-User-ID and X-User-Role headers if valid
// Otherwise, it returns 401 Unauthorized
func (c *TokenValidatorController) ValidateToken(w http.ResponseWriter, r *http.Request) {
        var tokenString string

        // Try extracting from cookie first
        cookie, err := r.Cookie("access_token")
        if err == nil && cookie.Value != "" {
                tokenString = cookie.Value
        } else {
                // Fallback to Authorization header
                authHeader := r.Header.Get("Authorization")
                if authHeader == "" {
                        http.Error(w, "Unauthorized: missing token", http.StatusUnauthorized)
                        return
                }
                parts := strings.Split(authHeader, " ")
                if len(parts) != 2 || parts[0] != "Bearer" {
                        http.Error(w, "Invalid token format", http.StatusUnauthorized)
                        return
                }
                tokenString = parts[1]
        }

        claims, err := c.jwtService.ValidateToken(tokenString)
        if err != nil {
                http.Error(w, "Invalid token", http.StatusUnauthorized)
                return
        }

        userID, _ := claims["sub"].(string)
        roleID, _ := claims["role"].(string)

        w.Header().Set("X-User-ID", userID)
        w.Header().Set("X-User-Role", roleID)
        w.WriteHeader(http.StatusOK)
}

type contextKey string
const (
        UserIDKey     contextKey = "userID"
        RoleIDKey     contextKey = "roleID"
        IsExternalKey contextKey = "isExternal"
)

// JWTService interface local para evitar ciclo de importación
type JWTService interface {
        ValidateToken(tokenString string) (jwt.MapClaims, error)
}

// AuthMiddleware verifica el token JWT en el header de la solicitud
func AuthMiddleware(jwtService JWTService) func(http.Handler) http.Handler {
        return func(next http.Handler) http.Handler {
                return http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) {
                        var tokenString string
                        
                        // Primero intentar extraer de la cookie "access_token"
                        cookie, err := r.Cookie("access_token")
                        if err == nil && cookie.Value != "" {
                                tokenString = cookie.Value
                        } else {
                                // Fallback: Authorization header
                                authHeader := r.Header.Get("Authorization")
                                if authHeader == "" {
                                        http.Error(w, "Unauthorized: missing token", http.StatusUnauthorized)
                                        return
                                }
                                parts := strings.Split(authHeader, " ")
                                if len(parts) != 2 || parts[0] != "Bearer" {
                                        http.Error(w, "Invalid token format", http.StatusUnauthorized)
                                        return
                                }
                                tokenString = parts[1]
                        }

                        claims, err := jwtService.ValidateToken(tokenString)
                        if err != nil {
                                http.Error(w, "Invalid token", http.StatusUnauthorized)
                                return
                        }

                        userID, _ := claims["sub"].(string)
                        roleID, _ := claims["role"].(string)
                        isExternal, _ := claims["is_external"].(bool)

                        ctx := context.WithValue(r.Context(), UserIDKey, userID)
                        ctx = context.WithValue(ctx, RoleIDKey, roleID)
                        ctx = context.WithValue(ctx, IsExternalKey, isExternal)
                        
                        next.ServeHTTP(w, r.WithContext(ctx))
                })
        }
}

// RequirePermissionMiddleware verifica si el usuario en el contexto tiene permisos
func RequirePermissionMiddleware(permissionService interface{
        CheckPermission(ctx context.Context, roleID, action, resource string) (bool, error)
}, requiredAction, requiredResource string) func(http.Handler) http.Handler {
        return func(next http.Handler) http.Handler {
                return http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) {
                        
                        // Extraer el UserID o RoleID del contexto (puesto por AuthMiddleware)
                        userIDVal := r.Context().Value(UserIDKey)
                        if userIDVal == nil {
                                http.Error(w, "Unauthorized", http.StatusUnauthorized)
                                return
                        }
                        roleIDVal := r.Context().Value(RoleIDKey)
                        if roleIDVal == nil {
                                http.Error(w, "Unauthorized: missing role", http.StatusUnauthorized)
                                return
                        }
                        roleID := roleIDVal.(string)

                        // Consultar BD para verificar si el rol tiene {requiredAction} sobre {requiredResource}
                        hasPermission, err := permissionService.CheckPermission(r.Context(), roleID, requiredAction, requiredResource)
                        
                        if err != nil || !hasPermission {
                                http.Error(w, "Forbidden: insufficient permissions", http.StatusForbidden)
                                return
                        }

                        next.ServeHTTP(w, r)
                })
        }
}

// RequireRole verifica que el usuario autenticado (cuyo RoleID está en el contexto)
// tenga uno de los roles permitidos.
func RequireRole(allowedRoles ...string) func(http.Handler) http.Handler {
        return func(next http.Handler) http.Handler {
                return http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) {
                        if len(allowedRoles) == 0 {
                                next.ServeHTTP(w, r)
                                return
                        }

                        roleIDVal := r.Context().Value(RoleIDKey)
                        if roleIDVal == nil {
                                http.Error(w, "Unauthorized: missing role in context", http.StatusUnauthorized)
                                return
                        }

                        userRole := roleIDVal.(string)

                        allowed := false
                        for _, role := range allowedRoles {
                                if userRole == role {
                                        allowed = true
                                        break
                                }
                        }

                        // Admin y dev tienen bypass global de roles
                        if userRole == "role_admin" || userRole == "role_dev" {
                                allowed = true
                        }

                        if !allowed {
                                http.Error(w, "Forbidden: insufficient role permissions", http.StatusForbidden)
                                return
                        }

                        next.ServeHTTP(w, r)
                })
        }
}

// SetupMiddlewares configures global middlewares for the application
func SetupMiddlewares(r *chi.Mux) {
        // CORS Configuration (Must be registered first to process OPTIONS preflight requests)
        allowedOriginsEnv := os.Getenv("CORS_ALLOWED_ORIGINS")
        var allowedOrigins []string
        if allowedOriginsEnv != "" {
                for _, o := range strings.Split(allowedOriginsEnv, ",") {
                        if trimmed := strings.TrimSpace(o); trimmed != "" {
                                allowedOrigins = append(allowedOrigins, trimmed)
                        }
                }
        }

        corsMiddleware := cors.New(cors.Options{
                AllowedOrigins:   allowedOrigins,
                AllowedMethods:   []string{"GET", "POST", "PUT", "PATCH", "DELETE", "OPTIONS"},
                AllowedHeaders:   []string{"Accept", "Authorization", "Content-Type", "X-CSRF-Token", "X-Requested-With"},
                ExposedHeaders:   []string{"Link", "Set-Cookie"},
                AllowCredentials: true,
                MaxAge:           300,
        })
        r.Use(corsMiddleware.Handler)

        // Standard chi middlewares
        r.Use(middleware.RequestID)
        r.Use(middleware.RealIP)
        r.Use(middleware.Recoverer)

        // Logging middleware adapted for JSON / Google Cloud Logging
        r.Use(func(next http.Handler) http.Handler {
                return http.HandlerFunc(func(w http.ResponseWriter, req *http.Request) {
                        slog.Info("Incoming Request",
                                slog.String("method", req.Method),
                                slog.String("url", req.URL.Path),
                                slog.String("ip", req.RemoteAddr),
                        )
                        next.ServeHTTP(w, req)
                })
        })

        // Security Headers (OWASP)
        r.Use(func(next http.Handler) http.Handler {
                return http.HandlerFunc(func(w http.ResponseWriter, req *http.Request) {
                        // Prevent MIME sniffing (A05)
                        w.Header().Set("X-Content-Type-Options", "nosniff")
                        // Prevent clickjacking (A05)
                        w.Header().Set("X-Frame-Options", "DENY")
                        // HSTS: enforce HTTPS for 1 year including subdomains (A02/A05)
                        w.Header().Set("Strict-Transport-Security", "max-age=31536000; includeSubDomains; preload")
                        // CSP: restrict resource origins to known domains (A03/A05)
                        w.Header().Set("Content-Security-Policy",
                                "default-src 'self'; "+
                                        "connect-src 'self' http://localhost:4200 https://ceibuenvivir.org https://testing.ceibuenvivir.pages.dev; "+
                                        "frame-ancestors 'none';")
                        // Prevent referrer leakage (A01)
                        w.Header().Set("Referrer-Policy", "strict-origin-when-cross-origin")
                        // Disable unnecessary browser features (A05)
                        w.Header().Set("Permissions-Policy", "geolocation=(), microphone=(), camera=()")
                        next.ServeHTTP(w, req)
                })
        })
}

func (r *companyRepository) FindByID(ctx context.Context, id string) (*models.Company, error) {
        var company models.Company
        if err := r.db.WithContext(ctx).First(&company, "company_id = ?", id).Error; err != nil {
                if errors.Is(err, gorm.ErrRecordNotFound) {
                        return nil, errors.New("empresa no encontrada")
                }
                return nil, err
        }
        return &company, nil
}

func (r *companyRepository) FindAll(ctx context.Context) ([]models.Company, error) {
        var companies []models.Company
        if err := r.db.WithContext(ctx).Find(&companies).Error; err != nil {
                return nil, err
        }
        return companies, nil
}

func (r *companyRepository) Create(ctx context.Context, company *models.Company) error {
        return r.db.WithContext(ctx).Transaction(func(tx *gorm.DB) error {
                defer func() {
                        if r := recover(); r != nil {
                                tx.Rollback()
                        }
                }()
                return tx.Create(company).Error
        })
}

func (r *companyRepository) Update(ctx context.Context, company *models.Company) error {
        return r.db.WithContext(ctx).Transaction(func(tx *gorm.DB) error {
                defer func() {
                        if r := recover(); r != nil {
                                tx.Rollback()
                        }
                }()
                return tx.Save(company).Error
        })
}

        return r.db.WithContext(ctx).Transaction(func(tx *gorm.DB) error {
                defer func() {
                        if r := recover(); r != nil {
                                tx.Rollback()
                        }
                }()
                return tx.Where("company_id = ?", id).Delete(&models.Company{}).Error
        })
}

// UpdateMyProfile dynamically updates the specific role table for the user
func (s *authService) UpdateMyProfile(ctx context.Context, roleID, userID string, data map[string]interface{}) error {
        var tableName string
        switch roleID {
        case "role_participant":
                tableName = "participants"
        case "role_company":
                tableName = "companies"
        case "role_teacher":
                tableName = "teachers"
        case "role_volunteer":
                tableName = "volunteers"
        default:
                return errors.New("rol no soporta perfil extra o es inválido")
        }

        // Remove unupdatable fields just in case
        delete(data, "user_id")
        delete(data, "approval_status") // security: they shouldn't auto-approve themselves
        delete(data, "approval_date")
        delete(data, "approved_by")
        delete(data, "created_at")
        delete(data, "updated_at")
        delete(data, "participant_id")
        delete(data, "company_id")
        delete(data, "teacher_id")
        delete(data, "volunteer_id")

        if len(data) == 0 {
                return nil // nothing to update
        }

        err := s.db.WithContext(ctx).Table(tableName).Where("user_id = ?", userID).Updates(data).Error
        if err != nil {
                return fmt.Errorf("error actualizando perfil: %w", err)
        }
        return nil
}

func (s *authService) Register(ctx context.Context, req *dto.RegisterRequest) (*dto.LoginResponse, error) {
        if req.Email == "" || req.Password == "" || req.Role == "" {
                return nil, errors.New("el correo electrónico, la contraseña y el rol son obligatorios")
        }

        if err := utils.ValidatePasswordComplexity(req.Password); err != nil {
                return nil, err
        }

        user, err := s.userRepo.FindByEmail(ctx, req.Email)
        if err == nil && user != nil {
                return nil, errors.New("ya existe una cuenta registrada con este correo electrónico")
        }

        roleID := req.Role
        if !strings.HasPrefix(roleID, "role_") {
                roleID = "role_" + roleID
        }

        hashedPassword, err := utils.HashPassword(req.Password)
        if err != nil {
                return nil, errors.New("error al procesar la contraseña")
        }

        user = &models.User{
                UserID:       "usr-" + utils.GenerateRandomString(10),
                FirstName:    req.FirstName,
                LastName:     req.LastName,
                Email:        req.Email,
                Phone:        req.Phone,
                PasswordHash: hashedPassword,
                RoleID:       roleID,
                Status:       "active",
        }

        err = s.db.WithContext(ctx).Transaction(func(tx *gorm.DB) error {
                if err := tx.Create(user).Error; err != nil {
                        return errors.New("error al guardar los datos del usuario")
                }
                if err := s.createRoleProfileTx(tx, user.UserID, roleID); err != nil {
                        return errors.New("error al crear el perfil asociado al rol seleccionado")
                }
                return nil
        })
        if err != nil {
                return nil, err
        }

        accessToken, refreshToken, err := s.jwtService.GenerateTokenPair(user.UserID, user.RoleID, false)
        if err != nil {
                return nil, errors.New("error al generar la sesión de autenticación")
        }

        if err := s.saveRefreshToken(ctx, user.UserID, refreshToken); err != nil {
                return nil, errors.New("error al guardar el estado de la sesión")
        }

        return &dto.LoginResponse{
                Token:        accessToken,
                RefreshToken: refreshToken,
                User:         user,
        }, nil
}

// CheckCredentials validates email and password and returns all accounts associated with the email.
// No token is issued at this step.
func (s *authService) CheckCredentials(ctx context.Context, req *dto.CheckCredentialsRequest) (*dto.CheckCredentialsResponse, error) {
        if req.Email == "" || req.Password == "" {
                return nil, errors.New("el correo electrónico y la contraseña son obligatorios")
        }

        users, err := s.userRepo.FindAllByEmail(ctx, req.Email)
        if err != nil {
                return nil, errors.New("no existe ninguna cuenta registrada con este correo electrónico")
        }

        // Validate password against the first account (all accounts with the same email share the same password hash)
        if !utils.CheckPasswordHash(req.Password, users[0].PasswordHash) {
                slog.Warn("security: failed login attempt", slog.String("event", "invalid_password"), slog.String("email_hash", utils.HashToken(req.Email)))
                return nil, errors.New("la contraseña ingresada es incorrecta")
        }

        roleNames := map[string]string{
                "role_admin":       "Responsable",
                "role_dev":         "Desarrollador",
                "role_participant": "Participante",
                "role_company":     "Empresa",
                "role_teacher":     "Docente",
                "role_volunteer":   "Voluntario",

// CheckGoogleCredentials validates a Firebase ID token and returns available accounts linked to the Google account.
// Google login requires that an account with the email exists AND has a matching google_auth_id linked.
func (s *authService) CheckGoogleCredentials(ctx context.Context, req *dto.GoogleLoginRequest) (resp *dto.CheckCredentialsResponse, err error) {
        defer func() {
                if r := recover(); r != nil {
                        err = fmt.Errorf("error crítico inesperado: %v", r)

        }()

        fbToken, err := s.firebaseService.VerifyIDToken(ctx, req.IDToken)
        if err != nil {
                return nil, errors.New("el token de autenticación de Google es inválido o expiró")
        }

        email, ok := fbToken.Claims["email"].(string)
        if !ok || email == "" {
                return nil, errors.New("no se pudo obtener la dirección de correo electrónico desde Google")
        }

        googleAuthID := strings.TrimSpace(req.GoogleID)
        if googleAuthID == "" {

        }

        matchingUsers, err := s.userRepo.FindAllByGoogleID(ctx, googleAuthID)

        }

        // 2. Si no se encuentra por google_auth_id, comprobar por correo electrónico para reportar error preciso
        if len(matchingUsers) == 0 {
                users, _ := s.userRepo.FindAllByEmail(ctx, email)
                if len(users) == 0 {
                        return nil, errors.New("no existe ninguna cuenta registrada con este correo electrónico. Debe registrarse primero")
                }
                slog.Warn("security: google login denied — google_auth_id not linked", slog.String("event", "google_auth_not_linked"), slog.String("email_hash", utils.HashToken(email)))
                return nil, errors.New("la cuenta de Google no está vinculada a este usuario. Debe vincularla desde la configuración de su cuenta")
        }

        roleNames := map[string]string{
                "role_admin":       "Responsable",
                "role_dev":         "Desarrollador",
                "role_participant": "Participante",
                "role_company":     "Empresa",
                "role_teacher":     "Docente",
                "role_volunteer":   "Voluntario",
        }

        accounts := make([]dto.AvailableAccount, 0, len(matchingUsers))
        for _, u := range matchingUsers {
                roleName := roleNames[u.RoleID]
                if roleName == "" {
                        roleName = u.RoleID
                }
                accounts = append(accounts, dto.AvailableAccount{
                        UserID:   u.UserID,
                        RoleID:   u.RoleID,

                })
        }

        googleUserIDs := make([]string, 0, len(matchingUsers))
        for _, u := range matchingUsers {
                googleUserIDs = append(googleUserIDs, u.UserID)
        }
        googleNonce, nonceErr := utils.GetNonceStore().Generate(googleUserIDs, 2*time.Minute)
        if nonceErr != nil {
                return nil, errors.New("error generando sesión temporal, intente nuevamente")
        }

        return &dto.CheckCredentialsResponse{Accounts: accounts, LoginNonce: googleNonce}, nil
}

// Login is step 2: given a user_id, role_id and a valid login nonce, issue the JWT token.
func (s *authService) Login(ctx context.Context, req *dto.LoginRequest) (*dto.LoginResponse, error) {
        if req.UserID == "" || req.RoleID == "" {
                return nil, errors.New("el identificador de usuario y el rol son obligatorios")
        }

        // Validate and consume the single-use login nonce (A04: prevents Step 2 bypass)
        if !utils.GetNonceStore().Consume(req.LoginNonce, req.UserID) {
                slog.Warn("security: login step 2 rejected — invalid or expired nonce",
                        slog.String("event", "invalid_login_nonce"),
                        slog.String("user_id", req.UserID),
                )
                return nil, errors.New("la sesión de verificación expiró o es inválida. Por favor, iniciá sesión nuevamente")
        }

        user, err := s.userRepo.FindByID(ctx, req.UserID)
        if err != nil {
                return nil, errors.New("el usuario especificado no fue encontrado")
        }

        if user.RoleID != req.RoleID {
                slog.Warn("security: role mismatch on login step 2", slog.String("event", "role_mismatch"), slog.String("user_id", req.UserID), slog.String("requested_role", req.RoleID))
                return nil, errors.New("el rol seleccionado no corresponde a este usuario")
        }

        statusLower := strings.ToLower(strings.TrimSpace(user.Status))
        if statusLower != "active" {
                slog.Warn("security: login attempt on blocked account", slog.String("event", "account_blocked"), slog.String("user_id", user.UserID), slog.String("role_id", user.RoleID))
                return &dto.LoginResponse{
                        User:       user,
                        UserStatus: "blocked",
                }, nil
        }

        approvalStatus := s.GetApprovalStatus(ctx, user.RoleID, user.UserID)
        if user.RoleID != "role_admin" && user.RoleID != "role_dev" {
                if approvalStatus != "active" && approvalStatus != "accepted" && approvalStatus != "pending" {
                        return nil, errors.New("el perfil aún no está aprobado")
                }
        }

        isExternal := false
        if user.RoleID == "role_teacher" {
                var isExt bool
                if err := s.db.WithContext(ctx).Table("teachers").Where("user_id = ?", user.UserID).Select("is_external").Scan(&isExt).Error; err == nil {
                        isExternal = isExt
                }
        }

        accessToken, refreshToken, err := s.jwtService.GenerateTokenPair(user.UserID, user.RoleID, isExternal)
        if err != nil {
                return nil, errors.New("error generando token")
        }

        if err := s.saveRefreshToken(ctx, user.UserID, refreshToken); err != nil {
                return nil, errors.New("error al guardar sesión")
        }

        return &dto.LoginResponse{
                Token:          accessToken,
                RefreshToken:   refreshToken,
                User:           user,
                ApprovalStatus: approvalStatus,
                UserStatus:     user.Status,
        }, nil
}

func (s *authService) UpdateMyPassword(ctx context.Context, userID, oldPassword, newPassword string) error {
        user, err := s.userRepo.FindByID(ctx, userID)
        if err != nil {
                return errors.New("usuario no encontrado")
        }

        if user.PasswordHash == "" {
                return errors.New("la cuenta no tiene contraseña configurada (quizás fue creada con Google)")
        }

        if err := bcrypt.CompareHashAndPassword([]byte(user.PasswordHash), []byte(oldPassword)); err != nil {
                return errors.New("la contraseña actual es incorrecta")
        }

        hashedPassword, err := bcrypt.GenerateFromPassword([]byte(newPassword), bcrypt.DefaultCost)
        if err != nil {
                return errors.New("error al encriptar la nueva contraseña")
        }

        user.PasswordHash = string(hashedPassword)
        if err := s.userRepo.Update(ctx, user); err != nil {
                return errors.New("error al guardar la nueva contraseña")
        }

        return nil
}

func (s *authService) LinkGoogleAccount(ctx context.Context, userID string, req *dto.LinkGoogleRequest) error {
        fbToken, err := s.firebaseService.VerifyIDToken(ctx, req.IDToken)
        if err != nil {
                return errors.New("el token de autenticación de Google es inválido o expiró")
        }

        googleAuthID := strings.TrimSpace(req.GoogleID)
        if googleAuthID == "" {
                googleAuthID = fbToken.UID
        }

        currentUser, err := s.userRepo.FindByID(ctx, userID)
        if err != nil {
                return errors.New("usuario no encontrado")
        }

        allUsers, _ := s.userRepo.FindAll(ctx)
        for _, u := range allUsers {
                if u.GoogleAuthID == googleAuthID && u.Email != currentUser.Email {
                        return errors.New("esta cuenta de Google ya está vinculada a otro usuario")
                }
        }

        users, err := s.userRepo.FindAllByEmail(ctx, currentUser.Email)
        if err != nil {
                return errors.New("error al obtener las cuentas del usuario")
        }

        for i := range users {
                users[i].GoogleAuthID = googleAuthID
                if err := s.userRepo.Update(ctx, &users[i]); err != nil {
                        return errors.New("error al vincular la cuenta de Google")
                }
        }

        return nil
}

func (s *authService) UnlinkGoogleAccount(ctx context.Context, userID string) error {
        currentUser, err := s.userRepo.FindByID(ctx, userID)
        if err != nil {
                return errors.New("usuario no encontrado")
        }

        users, err := s.userRepo.FindAllByEmail(ctx, currentUser.Email)
        if err != nil {
                return errors.New("error al obtener las cuentas del usuario")
        }

        hasPassword := false
        for _, u := range users {
                if strings.TrimSpace(u.PasswordHash) != "" {
                        hasPassword = true
                        break
                }
        }

        if !hasPassword {
                return errors.New("no podés desvincular la cuenta de Google porque no tenés una contraseña configurada")
        }

        for i := range users {
                users[i].GoogleAuthID = ""
                if err := s.userRepo.Update(ctx, &users[i]); err != nil {
                        return errors.New("error al desvincular la cuenta de Google")
                }
        }

        return nil
}

// NewFirebaseService initializes the Firebase app and returns the service
func NewFirebaseService() (FirebaseService, error) {
        ctx := context.Background()
        
        // Si GOOGLE_APPLICATION_CREDENTIALS no está seteado, Firebase intentará usar Application Default Credentials
        var app *firebase.App
        var err error
        
        credentialsEnv := os.Getenv("GOOGLE_APPLICATION_CREDENTIALS")
        credentialsJSON := os.Getenv("FIREBASE_CREDENTIALS_JSON")
        
        if credentialsJSON != "" {
                opt := option.WithCredentialsJSON([]byte(credentialsJSON))
                app, err = firebase.NewApp(ctx, nil, opt)
        } else if credentialsEnv != "" {
                // Check if it's a raw JSON string instead of a path
                if len(credentialsEnv) > 0 && credentialsEnv[0] == '{' {
                        // Borramos la variable de entorno para que el SDK base no intente abrirla como archivo (file name too long)
                        os.Unsetenv("GOOGLE_APPLICATION_CREDENTIALS")
                        opt := option.WithCredentialsJSON([]byte(credentialsEnv))
                        app, err = firebase.NewApp(ctx, nil, opt)
                } else {
                        opt := option.WithCredentialsFile(credentialsEnv)
                        app, err = firebase.NewApp(ctx, nil, opt)
                }
        } else {
                app, err = firebase.NewApp(ctx, nil)
        }

        if err != nil {
                return nil, fmt.Errorf("error initializing firebase app: %v", err)
        }

        authClient, err := app.Auth(ctx)
        if err != nil {
                return nil, fmt.Errorf("error getting firebase auth client: %v", err)
        }

        return &firebaseService{
                authClient: authClient,
        }, nil
}

// VerifyIDToken verifies a Firebase ID token and returns the token payload.
func (s *firebaseService) VerifyIDToken(ctx context.Context, idToken string) (*auth.Token, error) {
        token, err := s.authClient.VerifyIDToken(ctx, idToken)
        if err != nil {
                return nil, fmt.Errorf("invalid firebase token: %v", err)
        }
        return token, nil
}

// JWTService handles generation and verification of JWTs
type JWTService interface {
        GenerateTokenPair(userID string, role string, isExternal bool) (string, string, error)
        ValidateToken(tokenString string) (jwt.MapClaims, error)
}

type jwtService struct {
        accessSecret  []byte
        refreshSecret []byte
        accessTTL     time.Duration
        refreshTTL    time.Duration
}

func NewJWTService() JWTService {
        accessSecret := os.Getenv("JWT_SECRET")
        refreshSecret := os.Getenv("JWT_REFRESH_SECRET")

        insecureDefaults := map[string]bool{
                "":                       true,
                "fallback-access-secret":  true,
                "fallback-refresh-secret": true,
                "super-secret-access-key":  true,
                "super-secret-refresh-key": true,
        }

        if insecureDefaults[accessSecret] {
                fmt.Fprintln(os.Stderr, "FATAL: JWT_SECRET is not set or uses an insecure default. Set a strong secret in environment variables.")
                os.Exit(1)
        }
        if insecureDefaults[refreshSecret] {
                fmt.Fprintln(os.Stderr, "FATAL: JWT_REFRESH_SECRET is not set or uses an insecure default. Set a strong secret in environment variables.")
                os.Exit(1)
        }

        accessMin, _ := strconv.Atoi(os.Getenv("JWT_EXPIRATION_MINUTES"))
        if accessMin == 0 {
                accessMin = 15
        }

        refreshDays, _ := strconv.Atoi(os.Getenv("JWT_REFRESH_EXPIRATION_DAYS"))
        if refreshDays == 0 {
                refreshDays = 7
        }

        return &jwtService{
                accessSecret:  []byte(accessSecret),
                refreshSecret: []byte(refreshSecret),
                accessTTL:     time.Duration(accessMin) * time.Minute,
                refreshTTL:    time.Duration(refreshDays) * 24 * time.Hour,
        }
}

func (s *jwtService) GenerateTokenPair(userID string, role string, isExternal bool) (string, string, error) {
        // Access Token
        accessTokenClaims := jwt.MapClaims{
                "sub":         userID,
                "role":        role,
                "is_external": isExternal,
                "exp":         time.Now().Add(s.accessTTL).Unix(),
                "iat":  time.Now().Unix(),
                "type": "access",
        }
        accessToken := jwt.NewWithClaims(jwt.SigningMethodHS256, accessTokenClaims)
        accessString, err := accessToken.SignedString(s.accessSecret)
        if err != nil {
                return "", "", fmt.Errorf("failed to sign access token: %w", err)
        }

        // Refresh Token
        refreshString, err := utils.GenerateSecureToken(32)
        if err != nil {
                return "", "", fmt.Errorf("failed to generate refresh token: %w", err)
        }

        return accessString, refreshString, nil
}

func (s *jwtService) ValidateToken(tokenString string) (jwt.MapClaims, error) {
        secret := s.accessSecret
        expectedType := "access"

        token, err := jwt.Parse(tokenString, func(token *jwt.Token) (interface{}, error) {
                if _, ok := token.Method.(*jwt.SigningMethodHMAC); !ok {
                        return nil, fmt.Errorf("unexpected signing method: %v", token.Header["alg"])
                }
                return secret, nil
        })

        if err != nil {
                return nil, err
        }

        claims, ok := token.Claims.(jwt.MapClaims)
        if !ok || !token.Valid {
                return nil, errors.New("invalid token")
        }

        if claims["type"] != expectedType {
                return nil, errors.New("invalid token type")
        }

        return claims, nil
}

                Email:        req.Email,
                PasswordHash: string(hashedPassword),
                Phone:        req.Phone,
                Status:       "active",
                RoleID:       req.RoleID,
        }

        // 4. Guardar en Base de Datos (a través del repositorio con transacciones)
        if err := s.userRepo.Create(ctx, user); err != nil {
                return nil, err
        }

        return dto.ToUserResponse(user), nil

// HashPassword generates a bcrypt hash from the given password.
func HashPassword(password string) (string, error) {
        bytes, err := bcrypt.GenerateFromPassword([]byte(password), 14)
        return string(bytes), err
}

// CheckPasswordHash compares a bcrypt hashed password with its possible plaintext equivalent.
func CheckPasswordHash(password, hash string) bool {
        err := bcrypt.CompareHashAndPassword([]byte(hash), []byte(password))
        return err == nil
}

// GenerateSecureToken generates a random URL-safe base64 string of the specified byte length.
func GenerateSecureToken(length int) (string, error) {
        b := make([]byte, length)
        if _, err := rand.Read(b); err != nil {
                return "", err
        }
        return base64.URLEncoding.WithPadding(base64.NoPadding).EncodeToString(b), nil
}

// HashToken generates a SHA-256 hash of the given token string and returns it as a hex string.
func HashToken(token string) string {
        hash := sha256.Sum256([]byte(token))
        return hex.EncodeToString(hash[:])
}

import (
        crand "crypto/rand"
        "encoding/hex"
        "sync"
        "time"
)


// nonceEntry holds the set of valid user IDs for a login nonce and its expiry time.
type nonceEntry struct {
        allowedUserIDs map[string]struct{}
        expiresAt      time.Time
}

// NonceStore is an in-memory, thread-safe store for single-use login nonces.
type NonceStore struct {
        mu      sync.Mutex
        entries map[string]nonceEntry
}

var globalNonceStore = newNonceStore()

func newNonceStore() *NonceStore {
        ns := &NonceStore{entries: make(map[string]nonceEntry)}
        go ns.cleanupLoop()
        return ns
}

// GetNonceStore returns the singleton NonceStore instance.
func GetNonceStore() *NonceStore {
        return globalNonceStore
}

// Generate creates a new single-use nonce bound to the given user IDs, valid for ttl duration.
// Returns the opaque nonce string.
func (ns *NonceStore) Generate(userIDs []string, ttl time.Duration) (string, error) {
        raw := make([]byte, 32)
        if _, err := crand.Read(raw); err != nil {
                return "", err
        }
        nonce := hex.EncodeToString(raw)

        allowed := make(map[string]struct{}, len(userIDs))
        for _, id := range userIDs {
                allowed[id] = struct{}{}
        }

        ns.mu.Lock()
        ns.entries[nonce] = nonceEntry{
                allowedUserIDs: allowed,
                expiresAt:      time.Now().Add(ttl),
        }
        ns.mu.Unlock()

        return nonce, nil
}

// Consume validates that the nonce exists, has not expired, and that userID is in its allowed set.
// On success it deletes the nonce (single-use guarantee) and returns true.
func (ns *NonceStore) Consume(nonce, userID string) bool {
        ns.mu.Lock()
        defer ns.mu.Unlock()

        entry, ok := ns.entries[nonce]
        if !ok || time.Now().After(entry.expiresAt) {
                delete(ns.entries, nonce)
                return false
        }
        if _, allowed := entry.allowedUserIDs[userID]; !allowed {
                return false
        }

        delete(ns.entries, nonce)
        return true
}

// cleanupLoop removes expired nonces every 5 minutes to prevent memory growth.
func (ns *NonceStore) cleanupLoop() {
        ticker := time.NewTicker(5 * time.Minute)
        defer ticker.Stop()
        for range ticker.C {
                now := time.Now()
                ns.mu.Lock()
                for k, v := range ns.entries {
                        if now.After(v.expiresAt) {
                                delete(ns.entries, k)
                        }
                }
                ns.mu.Unlock()
        }
}

        "encoding/hex"

const hexCharset = "abcdefghijklmnopqrstuvwxyzABCDEFGHIJKLMNOPQRSTUVWXYZ0123456789"

// GenerateRandomString returns a cryptographically secure random alphanumeric string of length n.
func GenerateRandomString(n int) string {
        b := make([]byte, n)
        if _, err := rand.Read(b); err != nil {
                panic("utils: failed to generate random string: " + err.Error())
        }
        // Map each byte to a character in the charset using modulo
        result := make([]byte, n)
        for i, v := range b {
                result[i] = hexCharset[int(v)%len(hexCharset)]
        }
        return string(result)
}

// GenerateHexID generates a cryptographically secure random hex string (suitable for IDs).
func GenerateHexID(n int) string {
        b := make([]byte, n)
        if _, err := rand.Read(b); err != nil {
                panic("utils: failed to generate hex ID: " + err.Error())
        }
        return hex.EncodeToString(b)[:n]
}

var validate *validator.Validate

// ValidatePasswordComplexity checks that a password has >=8 chars, 1 uppercase, 1 lowercase, 1 special character.
func ValidatePasswordComplexity(password string) error {
        if len(password) < 8 {
                return errors.New("la contraseña debe tener al menos 8 caracteres")
        }

        var hasUpper, hasLower, hasSpecial bool
        for _, ch := range password {
                switch {
                case unicode.IsUpper(ch):
                        hasUpper = true
                case unicode.IsLower(ch):
                        hasLower = true
                case !unicode.IsLetter(ch) && !unicode.IsNumber(ch):
                        hasSpecial = true
                }
        }

        if !hasUpper {
                return errors.New("la contraseña debe incluir al menos una letra mayúscula")
        }
        if !hasLower {
                return errors.New("la contraseña debe incluir al menos una letra minúscula")
        }
        if !hasSpecial {
                return errors.New("la contraseña debe incluir al menos un carácter especial (ej: !@#$%^&*)")
        }
        return nil
}

// GetValidator returns a singleton instance of the validator.
func GetValidator() *validator.Validate {
        if validate == nil {
                validate = validator.New()

                // Custom Validation for Password
                _ = validate.RegisterValidation("password", func(fl validator.FieldLevel) bool {
                        return ValidatePasswordComplexity(fl.Field().String()) == nil
                })
        }
        return validate
}

// FormatValidationError formats the validator.ValidationErrors into a readable map
func FormatValidationError(err error) map[string]string {
        errorsMap := make(map[string]string)
        if validationErrors, ok := err.(validator.ValidationErrors); ok {
                for _, e := range validationErrors {
                        errorsMap[e.Field()] = e.Tag()
                }
        } else {
                errorsMap["request"] = "Invalid request format"
        }
        return errorsMap
}
